cmake_minimum_required(VERSION 3.20)
project(DcsVrPrefetchFix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_options(/W4 /WX /permissive- /DUNICODE /D_UNICODE /DWIN32_LEAN_AND_MEAN /Brepro)
add_link_options(/Brepro)
# Loaded by DCS as bin\dxgi2.dll through Cheeky's dxgi.dll loader; static CRT keeps it self-contained.
add_library(prefetch_fix SHARED prefetch_fix.cpp)
target_link_libraries(prefetch_fix PRIVATE psapi)
add_library(prefetch_probe SHARED probe.cpp)
add_executable(prefetch_fix_tests prefetch_fix_tests.cpp)
enable_testing()
add_test(NAME prefetch_fix COMMAND prefetch_fix_tests WORKING_DIRECTORY $<TARGET_FILE_DIR:prefetch_fix_tests>)
